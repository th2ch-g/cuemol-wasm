cmake_minimum_required(VERSION 3.20)
project(CueMolWasmLCMS C)
get_filename_component(ADAPTER_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(LCMS_SOURCE "${ADAPTER_ROOT}/.cache/deps/Little-CMS-lcms2.17")
file(GLOB LCMS_SOURCES "${LCMS_SOURCE}/src/*.c")
add_library(lcms2 STATIC ${LCMS_SOURCES})
target_include_directories(lcms2 PUBLIC "${LCMS_SOURCE}/include")
install(TARGETS lcms2 ARCHIVE DESTINATION lib)
install(FILES "${LCMS_SOURCE}/include/lcms2.h" "${LCMS_SOURCE}/include/lcms2_plugin.h" DESTINATION include)
